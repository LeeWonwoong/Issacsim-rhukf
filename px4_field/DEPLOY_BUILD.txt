 1. [GPU 서버, acsl] 모델 export → 패키지 → USB(라벨 00A3-D44E)
──────────────────────────────────────────────────────────────────────────────
  cd ~/projects/Issacsim-rhukf
  # (새 학습 결과가 있을 때) .pt → npz
  python3 px4_field/export_policy.py results/claudecodefortest/isaac_v7/swirlB_s42/final_model.pt px4_field/models/swirl_v5B_s42.npz
  # 패키지 갱신 (env·calibration·configs·px4_field 코드·models·매뉴얼 → ~/swirl_field_pkg)
  PKG=~/swirl_field_pkg bash px4_field/deploy_jetson.sh
  rm -rf ~/swirl_field_pkg/px4_field/__pycache__ ~/swirl_field_pkg/env/__pycache__
  ls ~/swirl_field_pkg/px4_field/models/
  # USB 마운트(서버는 자동 마운트가 안 됨) → 복사 → 안전 제거
  sudo mkdir -p /media/acsl/usb
  sudo mount -o uid=$(id -u),gid=$(id -g) /dev/sda1 /media/acsl/usb        # lsblk 로 sda1 인지 확인
  rm -rf /media/acsl/usb/swirl_field_pkg
  cp -r ~/swirl_field_pkg /media/acsl/usb/ && sync
  ls /media/acsl/usb/swirl_field_pkg/px4_field/models/
  sudo umount /media/acsl/usb                                              # ★ 뒤에 뽑는다
  #   (콘솔에 앉아 있으면: udisksctl mount -b /dev/sda1 → /media/acsl/00A3-D44E/ 에 복사 → udisksctl unmount -b /dev/sda1)

 4. [젯슨] 의존성 확인·문제 해결 (처음 한 번)
──────────────────────────────────────────────────────────────────────────────
  source /opt/ros/humble/setup.bash; source ~/colcon_ws/install/setup.bash
  python3 -c "import numpy, rclpy, yaml, pymavlink; import px4_msgs.msg; print('OK', numpy.__version__)"
  sudo usermod -aG dialout quad          # (한 번, 재로그인) FC USB /dev/ttyACM0 권한. 급하면 sudo chmod 666 /dev/ttyACM0
  # ▶ ModuleNotFoundError: pyulog   → pip3 install --no-deps pyulog   (ulog 판정용. f13 자체는 없어도 뜸)
  # ▶ numpy.core.multiarray failed to import → pip 가 numpy 2 를 올린 것:
  #      pip3 uninstall -y numpy ; python3 -c "import numpy; print(numpy.__version__, numpy.__file__)"  → 1.2x /usr/lib/...
  #      (없으면 sudo apt install python3-numpy). pip 로는 항상 --no-deps 또는 "numpy<2".
  # ▶ 그림 창이 안 뜸 → DISPLAY 없음: ssh -X 로 접속하거나 노트북에서 그린다(FLIGHT_RUN.txt §3)
